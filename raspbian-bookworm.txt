Installation process for raspbian bookworm

# https://linuxize.com/post/how-to-install-python-3-7-on-debian-9/ # alternative to pyenv, not tested

# Install pyenv
# https://github.com/pyenv/pyenv?tab=readme-ov-file
# Install Python build dependencies before attempting to install a new Python version.
# https://github.com/pyenv/pyenv?tab=readme-ov-file#d-install-python-build-dependencies


$ pyenv install 3.7.17
$ pyenv global 3.7.17
$ cd ~
$ git clone https://github.com/JohnMarzulli/categorical-sectional.git
$ sudo apt install rng-tools
$ sudo apt install haveged
$ sudo raspi-config # Run 'raspi-config' and enable the SPI bus under Advanced
$ cd ~/categorical-sectional
$ python3 setup.py develop # No sudo needed as running under venv 3.7.17 on user pi
$ pip install "urllib3<2" "requests<2.32" # fix for:     _TYPE_REDUCE_RESULT = tuple[typing.Callable[..., object], tuple[object, ...]]
