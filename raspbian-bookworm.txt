Installation process for raspbian bookworm

# https://linuxize.com/post/how-to-install-python-3-7-on-debian-9/ # alternative to pyenv, not tested

# Install pyenv
# https://github.com/pyenv/pyenv?tab=readme-ov-file

# Install Python build dependencies before attempting to install a new Python version.
# https://github.com/pyenv/pyenv?tab=readme-ov-file#d-install-python-build-dependencies

$ pyenv install 3.7.17
$ pyenv global 3.7.17
$ cd ~
$ git clone https://github.com/JohnMarzulli/categorical-sectional.git
$ sudo apt install rng-tools
$ sudo apt install haveged
$ sudo raspi-config # Run 'raspi-config' and enable the SPI bus under Advanced
$ cd ~/categorical-sectional
$ python3 setup.py develop # No sudo needed as running under venv 3.7.17 on user pi
$ pip install "urllib3<2" "requests<2.32" # fix for:     _TYPE_REDUCE_RESULT = tuple[typing.Callable[..., object], tuple[object, ...]]
$ python controller.py &

# MapConfig
$ cd ~/categorical-sectional/MapConfig
$ npm install
$ node /home/pi/categorical-sectional/MapConfig/build/index.js &

# led service
$ cat /lib/systemd/system/map-led.service 
[Unit]
Description=categorical-sectional map leds
After=syslog.target network.target

[Service]
User=pi
WorkingDirectory=/home/pi/categorical-sectional
ExecStart=/home/pi/.pyenv/versions/3.7.17/bin/python controller.py

Restart=always

[Install]
WantedBy=multi-user.target

# web config service
$ cat /lib/systemd/system/map-webconf.service 
[Unit]
Description=categorical-sectional Web MapConfig
Documentation=https://github.com/JohnMarzulli/categorical-sectional/tree/release
Documentation=https://nodesource.com/blog/running-your-node-js-app-with-systemd-part-1
After=network.target

[Service]
Type=simple
ExecStartPre=/bin/bash -c 'while ! /bin/nc -z localhost 8080 >/dev/null 2>&1; do /bin/sleep 10; done'
ExecStart=/usr/bin/node /home/pi/categorical-sectional/MapConfig/build/index.js
Restart=on-failure

SyslogIdentifier=mapconfig

[Install]
WantedBy=multi-user.target
