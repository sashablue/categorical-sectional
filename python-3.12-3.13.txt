pyenv install 3.XX.XX
pyenv global <3.XX.XX>

sudo apt install nodejs
sudo apt install npm
git clone https://github.com/sashablue/categorical-sectional.git
cd categorical-sectional
### git checkout python-3.12-compat
git checkout feature/update-weather-api
pip install -e .
sudo raspi-config # Enable the SPI bus
cd ~/categorical-sectional/MapConfig/
npm install
cd ~/categorical-sectional
wget https://davidmegginson.github.io/ourairports-data/airports.csv -O data/airports.csv # download each time to be up to date
### sed -i 's*cgi-bin/data/metar.php?ids=*api/data/metar?ids=*' data_sources/weather.py # API details: https://aviationweather.gov/help/upcoming/#api
echo "vm.swappiness = 1" | sudo tee -a /etc/sysctl.conf ; sudo sysctl -p
sudo raspi-config - Menu 1, Menu S5, Menu B1, Reboot # Optionally disable the Desktop GUI
# Enable log2ram and tmpfs # Optional, needs some RAM depending on the running app memory usage
# https://www.dzombak.com/blog/2024/04/pi-reliability-reduce-writes-to-your-sd-card/

# Testing
python controller.py &
sudo node /home/pi/categorical-sectional/MapConfig/build/index.js &

# LED service
$ cat /lib/systemd/system/map-led.service 
[Unit]
Description=categorical-sectional map leds
After=syslog.target network.target

[Service]
User=pi
WorkingDirectory=/home/pi/categorical-sectional
ExecStart=/home/pi/.pyenv/versions/3.7.17/bin/python controller.py
Restart=always

SyslogIdentifier=map-led

[Install]
WantedBy=multi-user.target

# Web config page service
$ cat /lib/systemd/system/map-webconf.service 
[Unit]
Description=categorical-sectional Web MapConfig
Documentation=https://github.com/JohnMarzulli/categorical-sectional/tree/release
Documentation=https://nodesource.com/blog/running-your-node-js-app-with-systemd-part-1
After=network.target

[Service]
Type=simple
ExecStartPre=/bin/bash -c 'while ! /bin/nc -z localhost 8080 >/dev/null 2>&1; do /bin/sleep 10; done'
ExecStart=/usr/bin/node /home/pi/categorical-sectional/MapConfig/build/index.js
Restart=on-failure

SyslogIdentifier=map-webconf

[Install]
WantedBy=multi-user.target
